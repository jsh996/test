CREATE OR REPLACE VIEW DEV_DATASERVICES_DB.CLEAN.SRC_VENDOR_D_VW AS

SELECT
    "VENDOR"."LEGACY_ID" AS "VENDOR_NUMBER",
    "VENDOR"."VENDOR_NAME" AS "VENDOR_NAME",

    CASE
        WHEN "VENDOR_SERVICE"."SERVICE_STATUS" = 'f' THEN 'I'
        WHEN "VENDOR_SERVICE"."SERVICE_STATUS" = 't' THEN 'A'
        ELSE "VENDOR_SERVICE"."SERVICE_STATUS"
    END AS "VENDOR_STATUS",

    CASE
        WHEN "VENDOR"."COUNTRY_CODE" = 'USA' THEN 1
        WHEN "VENDOR"."COUNTRY_CODE" = 'CAN' THEN 2
    END AS "SOURCE_SYSTEM_ID",

    "VENDOR_SERVICE"."GROUP_ID" AS "VENDOR_GROUP_NUMBER",

    "VENDOR_ADDRESS"."ADDRESS_1" AS "VENDOR_STREET_ADDRESS_1",
    "VENDOR_ADDRESS"."ADDRESS_2" AS "VENDOR_STREET_ADDRESS_2",
    "VENDOR_ADDRESS"."CITY" AS "VENDOR_CITY_NAME",
    "VENDOR_ADDRESS"."STATE_PROVINCE_CODE" AS "VENDOR_STATE_PROV_CODE",
    "VENDOR_ADDRESS"."POSTAL_CODE" AS "VENDOR_POSTAL_CODE",

    "VENDOR_CONTACT"."FIRST_NAME" || ' ' ||
    "VENDOR_CONTACT"."LAST_NAME" AS "VENDOR_CONTACT_NAME",

    "VENDOR_CONTACT"."PHONE" ||
        NVL2(
            CASE
                WHEN "VENDOR_CONTACT"."PHONE_EXTENSION" = ' '
                    THEN NULL
                ELSE "VENDOR_CONTACT"."PHONE_EXTENSION"
            END,
            ' x ' || "VENDOR_CONTACT"."PHONE_EXTENSION",
            NULL
        ) AS "VENDOR_CONTACT_PHONE",

    "VENDOR_CONTACT"."EMAIL"
        AS "VENDOR_CONTACT_E_MAIL_ADDRESS",

    CASE
        WHEN "VENDOR_SERVICE"."DIRECT_ORDER_VENDOR_FLAG"
             IN ('false', 'f')
            THEN '0'
        WHEN "VENDOR_SERVICE"."DIRECT_ORDER_VENDOR_FLAG"
             IN ('true', 't')
            THEN '1'
        ELSE NULL
    END AS "DIRECT_ORDER_IND",

    CASE
        WHEN "VENDOR_SERVICE"."NATIONAL_PURCHASE_FLAG"
             IN ('false', 'f')
            THEN '0'
        WHEN "VENDOR_SERVICE"."NATIONAL_PURCHASE_FLAG"
             IN ('true', 't')
            THEN '1'
        ELSE NULL
    END AS "NATIONAL_PURCH_AGREEMENT_IND",

    "VENDOR_SERVICE"."HOMENET_ID" AS "HOMENET_ID",

    "L_STATE_MASTER"."COUNTRY_CODE"
        AS "VENDOR_COUNTRY_CODE",

    CASE
        WHEN "VENDOR_SERVICE"."PREFERRED_VENDOR_FLAG"
             IN ('false', 'f')
            THEN '0'
        WHEN "VENDOR_SERVICE"."PREFERRED_VENDOR_FLAG"
             IN ('true', 't')
            THEN '1'
        ELSE NULL
    END AS "PREF_VENDOR_IND",

    "VENDOR"."DIF_LOAD_TSP"
        AS "VENDOR_DIF_LOAD_TSP",

    "VENDOR_SERVICE"."DIF_LOAD_TSP"
        AS "VENDOR_SERVICE_DIF_LOAD_TSP",

    "VENDOR_ADDRESS"."DIF_LOAD_TSP"
        AS "VENDOR_ADDRESS_DIF_LOAD_TSP",

    "L_STATE_MASTER"."DIF_LOAD_TSP"
        AS "L_STATE_MASTER_DIF_LOAD_TSP",

    "VENDOR_SERVICE_CONTACT"."DIF_LOAD_TSP"
        AS "VENDOR_SERVICE_CONTACT_DIF_LOAD_TSP",

    "VENDOR_CONTACT"."DIF_LOAD_TSP"
        AS "VENDOR_CONTACT_DIF_LOAD_TSP"

FROM DEV_DATASERVICES_DB.CLEAN.VENDOR_VENDOR
    AS "VENDOR"

LEFT OUTER JOIN
    DEV_DATASERVICES_DB.CLEAN.VENDOR_VENDOR_SERVICE
    AS "VENDOR_SERVICE"

    ON "VENDOR"."VENDOR_ID"
       = "VENDOR_SERVICE"."VENDOR_ID"

   AND "VENDOR"."LEGACY_ID"
       = "VENDOR_SERVICE"."LEGACY_ID"

   AND "VENDOR_SERVICE"."DIF_DELETE_FLG" = 0

LEFT OUTER JOIN
    DEV_DATASERVICES_DB.CLEAN.VENDOR_VENDOR_ADDRESS
    AS "VENDOR_ADDRESS"

    ON "VENDOR_SERVICE"."PHYSICAL_ADDRESS_ID"
       = "VENDOR_ADDRESS"."VENDOR_ADDRESS_ID"

   AND "VENDOR_ADDRESS"."DIF_DELETE_FLG" = 0

LEFT OUTER JOIN
    DEV_DATASERVICES_DB.CLEAN.L_STATE_MASTER_UNIQUE_VW
    AS "L_STATE_MASTER"

    ON "VENDOR_ADDRESS"."STATE_PROVINCE_CODE"
       = "L_STATE_MASTER"."STATE_PROV_CODE"

   AND "L_STATE_MASTER"."DIF_DELETE_FLG" = 0


/* =========================================================
   JOIN CONTACT
   ========================================================= */

LEFT OUTER JOIN
    DEV_DATASERVICES_DB.CLEAN.VENDOR_VENDOR_SERVICE_CONTACT
    AS "VENDOR_SERVICE_CONTACT"

    ON "VENDOR_SERVICE"."VENDOR_SERVICE_ID"
       = "VENDOR_SERVICE_CONTACT"."VENDOR_SERVICE_ID"

   AND "VENDOR_SERVICE_CONTACT"."DIF_DELETE_FLG" = 0

   AND EXISTS
   (
       SELECT 1
       FROM DEV_DATASERVICES_DB.CLEAN.VENDOR_VENDOR_SERVICE_CONTACT
           AS "VENDOR_SERVICE_CONTACT_ROLE"

       WHERE "VENDOR_SERVICE_CONTACT_ROLE"."CONTACT_ID"
             = "VENDOR_SERVICE_CONTACT"."CONTACT_ID"

         AND "VENDOR_SERVICE_CONTACT_ROLE"."ROLE_ID" = 1

         AND "VENDOR_SERVICE_CONTACT_ROLE"."DIF_DELETE_FLG" = 0
   )

LEFT OUTER JOIN
    DEV_DATASERVICES_DB.CLEAN.VENDOR_VENDOR_CONTACT
    AS "VENDOR_CONTACT"

    ON "VENDOR_SERVICE_CONTACT"."CONTACT_ID"
       = "VENDOR_CONTACT"."CONTACT_ID"

   AND "VENDOR_CONTACT"."DIF_DELETE_FLG" = 0


/* =========================================================
   FILTER ACTIVE
   ========================================================= */

WHERE "VENDOR"."DIF_DELETE_FLG" = 0;
